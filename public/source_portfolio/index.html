cacheManager.ts // 15-Minute Node-Cache Strategy TTL: 900s (15 min) 
import NodeCache from 'node-cache';
const guildCache = new NodeCache({ stdTTL: 900, checkperiod: 120 });

export async function fetchUserGuilds(userId: string, token: string) {
  const cached = guildCache.get(userId);
  if (cached) return { data: cached, source: 'cache_hit' };

  try {
    const guilds = await discordApi.getGuilds(token);
    guildCache.set(userId, guilds);
    return { data: guilds, source: 'discord_api' };
  } catch (err: any) {
    if (err.status === 429 && cached) {
      logger.warn(`[429 RateLimit] Fallback to cached guilds for ${userId}`);
      return { data: cached, source: 'graceful_fallback' };
    }
    throw err;
  }
}
Fault Tolerance: Guarantees continuous dashboard responsiveness even when Discord API enforces 429 throttling on high-volume user refreshes.
